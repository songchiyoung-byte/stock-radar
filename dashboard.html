async function fetchRealtimePrices() { if (!isRealtimeRunning || stockData.length === 0) return; const tickers = stockData.map(s => s.ticker).filter(Boolean); // 1순위: 백엔드 API 호출 (Render 서버) try { const controller = new AbortController(); const timeoutId = setTimeout(() => controller.abort(), 6000); // 타임아웃을 6초로 상향 const res = await fetch(`/api/prices?tickers=${tickers.join(',')}`, { signal: controller.signal }); clearTimeout(timeoutId); if (res.ok) { const data = await res.json(); if (data.success && data.prices) { isBackendConnected = true; const statusEl = document.getElementById('liveStatusText'); if (statusEl && isRealtimeRunning) { statusEl.textContent = '🟢 증권 거래소 실시간 연동 (Live)'; statusEl.className = 'font-medium text-emerald-400'; } tickers.forEach(ticker => { const info = data.prices[ticker]; if (info && info.success && info.price > 0) { updateStockPriceInRuntime(ticker, info.price, info.diff, info.rate); } }); updateTimestamp(); return; } } } catch (err) { console.warn('Backend price fetch error:', err); } // 백엔드 실패 시 상태 표시 (임의로 가격을 변형하지 않고 직전 시세 유지) const statusEl = document.getElementById('liveStatusText'); if (statusEl && isRealtimeRunning) { statusEl.textContent = '⚠️ 시세 서버 응답 대기 중'; statusEl.className = 'font-medium text-amber-400'; } updateTimestamp(); }
